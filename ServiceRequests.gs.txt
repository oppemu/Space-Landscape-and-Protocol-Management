// ดึงข้อมูลการขอใช้พื้นที่ส่วนกลางทั้งหมดสำหรับ Admin (ชีต Requests แมปตรงตามภาพจริง)
function handleGetAllAdminRequests(e, ss) {
  var requestSheet = ss.getSheetByName("Requests") || ss.getActiveSheet();
  if (!requestSheet) return { success: false, message: "Sheet 'Requests' not found" };

  var displayData = requestSheet.getDataRange().getDisplayValues();
  var results = [];

  for (var i = 1; i < displayData.length; i++) {
    var row = displayData[i];
    if (!row[1] && !row[0]) continue; // เช็คจากคอลัมน์ B (RequestID) หรือ A (Timestamp)

    results.push({
      id: row[1] || row[0] || "",          // B: RequestID (เช่น OPPE0001)
      rowIndex: i + 1,
      timestamp: row[0] || "",            // A: Timestamp
      requestDate: row[2] || "",          // C: RequestDate
      applicantName: ((row[3] ? row[3] + " " : "") + (row[4] || "")).trim(), // D + E: ชื่อผู้ขอใช้
      faculty: row[5] || "",              // F: คณะ
      department: row[6] || "",           // G: หน่วยงาน
      phone: cleanPhone(row[7]),          // H: เบอร์โทร
      email: row[8] || "",                // I: อีเมล
      pinCode: row[9] || "",              // J: pinCode
      eventName: row[10] || "",           // K: ActivityName (ชื่องาน/กิจกรรม)
      startDate: row[14] || "",           // O: StartDate
      startTime: cleanDateTime(row[15]),  // P: StartTime
      endDate: row[16] || "",             // Q: EndDate
      endTime: cleanDateTime(row[17]),    // R: EndTime
      location: row[18] || "",            // S: Location (สถานที่ส่วนกลาง)
      originalFee: row[25] || "",         // Z: จำนวนเงินเต็ม
      taxAmount: row[26] || "",           // AA: ยกเว้น %
      waivedAmount: row[27] || "",        // AB: ค่าบำรุงสุทธิ
      docNumber: row[28] || "",           // AC: เลขที่หนังสือ
      docDate: row[29] || "",             // AD: ลงวันที่หนังสือ
      payDate: row[30] || "",             // AE: วันที่จ่ายเงินค่าบริการ
      receiptNo: row[31] || "",           // AF: เลขที่ใบเสร็จค่าบริการ
      depositDate: row[32] || "",         // AG: วันที่จ่ายเงินประกัน
      depositReceiptNo: row[33] || "",    // AH: เลขที่ใบเสร็จเงินประกัน
      depositAmount: row[34] || "",       // AI: จำนวนเงินประกัน
      status: row[35] || "Pending"        // AJ: สถานะ
    });
  }

  return { success: true, records: results.reverse() };
}

// อัปเดตข้อมูลเจ้าหน้าที่สำหรับคำขอใช้พื้นที่ส่วนกลาง (ชีต Requests)
function handleUpdateAdminRequest(params, ss) {
  var requestSheet = ss.getSheetByName("Requests") || ss.getActiveSheet();
  if (!requestSheet) return { success: false, message: "Sheet 'Requests' not found" };

  var targetId = params.requestId ? params.requestId.trim().toUpperCase() : "";
  var data = requestSheet.getDataRange().getDisplayValues();
  var rowIndex = -1;

  for (var i = 1; i < data.length; i++) {
    // เช็ค RequestID ที่คอลัมน์ B (Index 1)
    if (data[i][1] && String(data[i][1]).trim().toUpperCase() === targetId) {
      rowIndex = i + 1;
      break;
    }
  }

  if (rowIndex === -1) return { success: false, message: "ไม่พบรายการที่ต้องการแก้ไข" };

  // บันทึกลงคอลัมน์ Z (26) ถึง AJ (36)
  requestSheet.getRange(rowIndex, 26).setValue(params.originalFee || "");
  requestSheet.getRange(rowIndex, 27).setValue(params.taxAmount || "");
  requestSheet.getRange(rowIndex, 28).setValue(params.waivedAmount || "");
  requestSheet.getRange(rowIndex, 29).setValue(params.docNumber || "");
  requestSheet.getRange(rowIndex, 30).setValue(params.docDate || "");
  requestSheet.getRange(rowIndex, 31).setValue(params.payDate || "");
  requestSheet.getRange(rowIndex, 32).setValue(params.receiptNo || "");
  requestSheet.getRange(rowIndex, 33).setValue(params.depositDate || "");
  requestSheet.getRange(rowIndex, 34).setValue(params.depositReceiptNo || "");
  requestSheet.getRange(rowIndex, 35).setValue(params.depositAmount || "");
  requestSheet.getRange(rowIndex, 36).setValue(params.status || "Pending");

  return { success: true, message: "อัปเดตข้อมูลเรียบร้อยแล้ว!" };
}
